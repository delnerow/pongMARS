    addi $sp, $sp, -4
    sw $ra, 0($sp)
    
    addi $s0,$a0,0     # coluna 0-2
    addi $s1,$a1,0     # linha 0-5
	
    # linha 0
    jal DrawDot
    addi $a0,$s0,1
    move $a1, $s1
     jal DrawDot
    addi $a0,$s0,2
    move $a1, $s1
    jal DrawDot
    # linhas 1–4
    addi $a1,$s1,1
    draw0_loop1:
        # coluna 0
        addi $a0,$s0,0
         jal DrawDot
        # coluna 2
        addi $a0,$s0,2
        jal DrawDot

        addi $t2,$zero,5
        bne $a1,$t2,draw0_loop1

    # linha 5
    addi $a0,$s0,0
    addi $a1,$s1,5
    jal DrawDot
    addi $a0,$s0,1 
     jal DrawDot
    addi $a0,$s0,2
    jal DrawDot
    lw $ra, 0($sp)
    addi $sp, $sp, 4
    jr $ra
